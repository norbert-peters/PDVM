"""

# ========================================
# HANDLER-TYPEN ÜBERSICHT
# ========================================

"""
V3.1 workspace_pipeline() unterstützt 3 Modi:

1. NORMAL (skip_clear=False, default):
   - Arbeitsbereich wird geleert
   - Neue Inhalte werden eingefügt
   - Use Case: Views, Dialogs, Menü wechseln

2. UI-TOGGLE (skip_clear=True):
   - Arbeitsbereich bleibt unverändert
   - Nur UI-Element wird getoggelt
   - Use Case: Menü ein/aus, Stichtagsbar ein/aus

3. WELCOME (kein command_func):
   - Arbeitsbereich wird geleert
   - Willkommensbildschirm wird angezeigt
   - Use Case: Zurück zu Startmenü, leerer Arbeitsbereich
"""

# ========================================
# BEISPIEL 1: ZURÜCK ZUM STARTMENÜ
# ========================================

def handler_return_to_startmenu(self):
    """
    🏠 Lädt Startmenü + zeigt Willkommensbildschirm
    
    EFFEKT:
    - Arbeitsbereich wird geleert (View/Dialog verschwindet)
    - Startmenü wird geladen
    - Willkommensbildschirm erscheint
    """
    def _load_startmenu():
        # Startmenü laden
        self.menu_handler.load_startmenu()
        self._show_welcome_message()
    
    # skip_clear=False (default) → Arbeitsbereich leeren
    self.workspace_pipeline(command_func=_load_startmenu)


# ========================================
# BEISPIEL 2: MENÜ EIN/AUS
# ========================================

def handler_toggle_menu(self):
    """
    🔄 Toggle Menü-Sichtbarkeit
    
    EFFEKT:
    - Menü wird ein/ausgeschaltet
    - Arbeitsbereich bleibt unverändert (View bleibt sichtbar!)
    - Stichtagsbar bleibt sichtbar
    """
    def _toggle_menu():
        current_visible = self.vertical_menu_container.isVisible()
        self.vertical_menu_container.setVisible(not current_visible)
        self._menu_visible = not current_visible
    
    # skip_clear=True → Arbeitsbereich NICHT leeren!
    self.workspace_pipeline(command_func=_toggle_menu, skip_clear=True)


# ========================================
# BEISPIEL 3: STICHTAGSBAR EIN/AUS
# ========================================

def handler_toggle_stichtagsbar(self):
    """
    🔄 Toggle Stichtagsbar-Sichtbarkeit
    
    EFFEKT:
    - Stichtagsbar wird ein/ausgeschaltet
    - Arbeitsbereich bleibt unverändert (View bleibt sichtbar!)
    - Menü bleibt sichtbar
    """
    def _toggle_stichtagsbar():
        current_visible = self.stichtag_bar.isVisible()
        self.stichtag_bar.setVisible(not current_visible)
    
    # skip_clear=True → Arbeitsbereich NICHT leeren!
    self.workspace_pipeline(command_func=_toggle_stichtagsbar, skip_clear=True)


# ========================================
# BEISPIEL 4: VIEW ÖFFNEN
# ========================================

def handler_open_view(self, view_guid):
    """
    📊 Öffnet View im Arbeitsbereich
    
    EFFEKT:
    - Arbeitsbereich wird geleert (alte View verschwindet)
    - Neue View wird geladen
    - View erscheint im Arbeitsbereich
    """
    def _open_view():
        # View erstellen und in workspace_layout einfügen
        view_widget = self._create_view(view_guid)
        self.workspace_layout.addWidget(view_widget)
    
    # skip_clear=False (default) → Arbeitsbereich leeren
    self.workspace_pipeline(command_func=_open_view)


# ========================================
# BEISPIEL 5: DIALOG ÖFFNEN
# ========================================

def handler_open_dialog(self, frame_guid):
    """
    📝 Öffnet Dialog im Arbeitsbereich
    
    EFFEKT:
    - Arbeitsbereich wird geleert (alte View verschwindet)
    - Dialog wird geladen
    - Dialog erscheint im Arbeitsbereich
    """
    def _open_dialog():
        # Dialog erstellen und in workspace_layout einfügen
        dialog_widget = self._create_dialog(frame_guid)
        self.workspace_layout.addWidget(dialog_widget)
    
    # skip_clear=False (default) → Arbeitsbereich leeren
    self.workspace_pipeline(command_func=_open_dialog)


# ========================================
# BEISPIEL 6: MENÜ WECHSELN
# ========================================

def handler_load_menu(self, menu_guid):
    """
    📋 Lädt neues Menü + zeigt Willkommensbildschirm
    
    EFFEKT:
    - Arbeitsbereich wird geleert (View verschwindet)
    - Neues Menü wird geladen
    - Willkommensbildschirm erscheint
    """
    def _load_menu():
        # Menü laden
        self.menu_handler.load_menu(menu_guid)
        # Kein Widget zurückgeben = Willkommensbildschirm erscheint
    
    # skip_clear=False (default) → Arbeitsbereich leeren + Welcome zeigen
    self.workspace_pipeline(command_func=_load_menu)


# ========================================
# BEISPIEL 7: FEHLER ANZEIGEN
# ========================================

def handler_show_error(self, error_message):
    """
    ❌ Zeigt Fehlermeldung im Arbeitsbereich
    
    EFFEKT:
    - Arbeitsbereich wird geleert
    - Fehlermeldung wird angezeigt (rotes Widget)
    """
    # error_message Parameter → Arbeitsbereich automatisch geleert
    self.workspace_pipeline(error_message=error_message)

MERKE:
- UI-Toggles → skip_clear=True
- Content-Wechsel → skip_clear=False

"""
So wird die Pipeline im Menu-Handler integriert:

# v3_menu_handler.py
def execute_command(self, command_data: dict):
    '''Führt Menü-Command aus'''
    handler_name = command_data.get('handler')
    handler_args = command_data.get('handler_args', {})
    
    # Hole Handler-Funktion aus main_app
    if hasattr(self.main_app, handler_name):
        handler_func = getattr(self.main_app, handler_name)
        
        # Führe Handler aus (entscheidet selbst über skip_clear)
        return handler_func(**handler_args)
    else:
        logger.error(f"Handler nicht gefunden: {handler_name}")
        return False
"""

"""
✅ VORTEILE DER skip_clear LÖSUNG:

1. EINFACH: Ein Boolean-Parameter steuert Verhalten
2. LINEAR: Keine komplexe Logik, klarer Ablauf
3. VERSTÄNDLICH: Name erklärt sich selbst
4. FLEXIBEL: Jeder Handler entscheidet selbst
5. RÜCKWÄRTSKOMPATIBEL: Default False = altes Verhalten
6. ERWEITERBAR: Weitere Flags möglich (z.B. skip_welcome)

✅ USE CASES ABGEDECKT:

- ✅ View öffnen → skip_clear=False (Arbeitsbereich leeren)
- ✅ Dialog öffnen → skip_clear=False (Arbeitsbereich leeren)
- ✅ Menü wechseln → skip_clear=False (Arbeitsbereich leeren + Welcome)
- ✅ Zurück zu Startmenü → skip_clear=False (Arbeitsbereich leeren + Welcome)
- ✅ Menü ein/aus → skip_clear=True (Arbeitsbereich behalten!)
- ✅ Stichtagsbar ein/aus → skip_clear=True (Arbeitsbereich behalten!)
- ✅ Fehler anzeigen → skip_clear=False (Arbeitsbereich leeren)

🎉 PIPELINE EINFACH & MÄCHTIG!
"""
