"""
🎯 V3.1 HANDLER SKIP_CLEAR ÜBERSICHT

Dieses Dokument zeigt, welche Handler skip_clear=True verwenden
und warum das so entschieden wurde.
"""

# ========================================
# HANDLER MIT SKIP_CLEAR=TRUE
# ========================================

"""
Diese Handler schalten NUR UI-Elemente um, ohne den Arbeitsbereich zu leeren.
Der aktuelle Inhalt (View/Dialog) bleibt sichtbar!
"""

# 1. handler_toggle_menu.py
SKIP_CLEAR = True
# GRUND: Menü ein/aus → View bleibt sichtbar
# EFFEKT: Nur Menü-Container wird getoggelt

# 2. handler_toggle_stichtagsbar.py (noch zu erstellen)
SKIP_CLEAR = True
# GRUND: Stichtagsbar ein/aus → View bleibt sichtbar
# EFFEKT: Nur Stichtagsbar wird getoggelt


# ========================================
# HANDLER MIT SKIP_CLEAR=FALSE (default)
# ========================================

"""
Diese Handler laden neuen Content oder wechseln die Ansicht.
Der Arbeitsbereich wird geleert und neu befüllt.
"""

# 1. handler_open_view.py
SKIP_CLEAR = False
# GRUND: View öffnen → Arbeitsbereich leeren, neue View zeigen
# EFFEKT: Alte View verschwindet, neue View erscheint

# 2. handler_open_start_menu.py
SKIP_CLEAR = False
# GRUND: Startmenü → Arbeitsbereich leeren, Welcome zeigen
# EFFEKT: Aktuelle View verschwindet, Willkommensbildschirm erscheint

# 3. handler_show_dialog.py
SKIP_CLEAR = False
# GRUND: Dialog öffnen → Arbeitsbereich leeren, Dialog zeigen
# EFFEKT: Aktuelle View verschwindet, Dialog erscheint

# 4. handler_open_app_menu.py
SKIP_CLEAR = False
# GRUND: Menü wechseln → Arbeitsbereich leeren, Welcome zeigen
# EFFEKT: Aktuelle View verschwindet, neues Menü + Welcome

# 5. handler_show_info.py
SKIP_CLEAR = False (oder True - abhängig von Anforderung)
# GRUND: Info-Dialog zeigen → Kann über View liegen ODER View ersetzen
# EMPFEHLUNG: Popup-Dialog statt Arbeitsbereich verwenden

# 6. handler_show_help.py
SKIP_CLEAR = False (oder True - abhängig von Anforderung)
# GRUND: Hilfe zeigen → Kann über View liegen ODER View ersetzen
# EMPFEHLUNG: Popup-Dialog statt Arbeitsbereich verwenden


# ========================================
# SPEZIALFÄLLE
# ========================================

"""
Diese Handler haben besondere Anforderungen
"""

# 1. handler_reload_view.py
SKIP_CLEAR = False
# GRUND: View neu laden → Arbeitsbereich leeren, View neu erstellen
# HINWEIS: Alternativ könnte View.refresh() verwendet werden (skip_clear=True)

# 2. handler_logout.py
SKIP_CLEAR = False
# GRUND: Logout → Arbeitsbereich leeren, zur Login-Seite
# HINWEIS: Komplett neue Anwendung, nicht nur Arbeitsbereich

# 3. handler_execute_python.py
SKIP_CLEAR = False (oder True - abhängig)
# GRUND: Python-Code ausführen → Ergebnis anzeigen
# EMPFEHLUNG: 
#   - Wenn Ergebnis in View: skip_clear=False (View ersetzen)
#   - Wenn nur Ausführung: skip_clear=True (View bleibt)


# ========================================
# ENTSCHEIDUNGSBAUM
# ========================================

"""
WIE ENTSCHEIDE ICH SKIP_CLEAR FÜR EINEN HANDLER?

FRAGE 1: Wird der Arbeitsbereich verändert?
  │
  ├─ NEIN → skip_clear=True
  │         (Nur UI-Toggle, Einstellungen, etc.)
  │         Beispiele: Menü ein/aus, Theme wechseln, Vollbild toggle
  │
  └─ JA → Weiter zu FRAGE 2

FRAGE 2: Soll alter Content verschwinden?
  │
  ├─ JA → skip_clear=False
  │        (View/Dialog ersetzen)
  │        Beispiele: View öffnen, Dialog öffnen, Startmenü
  │
  └─ NEIN → skip_clear=True
            (Content bleibt, nur refresh)
            Beispiele: View refresh (ohne Neuerstellen)
"""

"""

   - Stichtag ändern → Content wird refreshed (nicht neu erstellt)
   
5. handler_save_layout.py
   - Layout speichern → Content bleibt

   - Zoom ändern → Content bleibt
